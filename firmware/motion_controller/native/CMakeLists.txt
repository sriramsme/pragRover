cmake_minimum_required(VERSION 3.16)
project(prag_motion_native LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
add_executable(control_tests control_tests.c ../core/control.c)
target_include_directories(control_tests PRIVATE ../core)
target_compile_features(control_tests PRIVATE c_std_11)
target_compile_options(control_tests PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
option(ROVER_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(ROVER_SANITIZE)
    target_compile_options(control_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(control_tests PRIVATE -fsanitize=address,undefined)
endif()
target_link_libraries(control_tests PRIVATE m)
add_test(NAME control_core COMMAND control_tests)

# Optional parser verification against the exact SDK cJSON implementation.
if(DEFINED IDF_PATH)
    add_executable(wire_tests wire_tests.c ../core/wire.c
        "${IDF_PATH}/components/json/cJSON/cJSON.c")
    target_include_directories(wire_tests PRIVATE ../core "${IDF_PATH}/components/json/cJSON")
    target_compile_features(wire_tests PRIVATE c_std_11)
    target_compile_options(wire_tests PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    if(ROVER_SANITIZE)
        target_compile_options(wire_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(wire_tests PRIVATE -fsanitize=address,undefined)
    endif()
    target_link_libraries(wire_tests PRIVATE m)
    add_test(NAME wire_parser COMMAND wire_tests)
endif()
